⛰️ Topographic Overlay for valheim-map.world

A bookmarklet that adds a height-shaded topographic map with contour lines on top of the Valheim World Generator — pannable, zoomable, and fully client-side.

How to use it

  1. Go to valheim-map.world and load or generate a world as usual.
  2. Click the Topo Overlay bookmark you just installed.
  3. A small mountain icon appears near the top-right of the page, just below the ? help button.
  4. Click the mountain icon to open the overlay panel.

Inside the panel:

  • Drag the panel by its header to move it, or drag its bottom-right corner to resize it.
  • Click-drag inside the map to pan, scroll to zoom.
  • Show Contour Lines toggles elevation contour lines, anchored at sea level (30m).
  • Preset buttons choose contour spacing: 5 / 10 / 15 / 20 / 25 meters.
  • Label Density and Label Opacity sliders tune how contour labels are drawn.
  • Refresh re-scans if you generate a new seed and the auto-refresh doesn't catch it.
  • Closing and reopening the panel keeps your last pan/zoom position.

What this actually does

This bookmarklet reads the world's height data directly out of the page's own WebAssembly memory (the same data the site already generated to render its own map) and draws a separate height-shaded overlay from it. It does not modify the site, does not send any data anywhere, and does not persist anything after you close the tab.

// ===========================================================================
// Valheim Map World - Topographic Overlay
//
// A bookmarklet/userscript that adds a topographic (height-shaded + contour
// line) overlay to https://valheim-map.world/, reading world height data
// directly out of the site's Unity WebGL heap. Does not modify or interact
// with the site's own rendering; adds a separate floating panel with its
// own pan/zoom.
//
// Usage: see README / install instructions. Click the mountain icon (added
// near the top-right, below the "?" help button) to toggle the panel.
//
// This file is safe to read top-to-bottom before running - it only reads
// from window.gameInstance.Module.HEAPF32 (the page's own WASM memory) and
// adds new DOM elements; it does not send any data anywhere or modify the
// underlying map/site behavior.
// ===========================================================================
(function () {
  'use strict';

  // ---------------------------------------------------------------------
  // CONFIG - tweak these freely, they drive all coloring/contour behavior
  // ---------------------------------------------------------------------
  const CONFIG = {
    seaLevel: 30,
    // Land color stops: [height, [r,g,b]]. Interpolated smoothly between
    // consecutive stops. Heights >= 150 fade from white toward lightBlue
    // at the map's actual max height.
    landStops: [
      [30, [42, 80, 52]],     // dark pine forest green
      [45, [176, 168, 112]],  // muted tan-yellow
      [60, [188, 149, 101]],  // muted orange-tan
      [75, [149, 116, 87]],   // muted brown
      [90, [142, 140, 133]],  // muted grey
      [115, [223, 220, 213]], // muted off-white
      [150, [223, 220, 213]], // off-white (end of plateau)
    ],
    lightBlueColor: [140, 178, 222], // more saturated blue "hot star" tint past 150 -> max
    water: {
      voidColor: [0, 0, 0],
      voidThreshold: -390,   // at/below this -> pure black void
      deepFloor: -50,        // below this, clamp to deepColor
      deepColor: [24, 34, 49],
      midColor: [54, 80, 105],      // color exactly at height 0
      shallowColor: [118, 148, 163], // color just below sea level (29)
    },
    contour: {
      spacingPresets: [5, 10, 15, 20, 25],
      defaultSpacing: 15,
      lineColor: 'rgba(0,0,0,0.55)',
      labelFillColor: '#ffffff',
      labelStrokeColor: '#000000',
      defaultLabelOpacity: 0.9,
      // "density" is 1 (sparse) - 10 (dense); mapped to target screen px between labels
      defaultLabelDensity: 5,
      densityToPxRange: [400, 50], // [px at density=1, px at density=10]
    },
  };

  // ---------------------------------------------------------------------
  // Cleanup any previous injection (safe to re-run this script)
  // ---------------------------------------------------------------------
  document.getElementById('topomap-toggle-btn')?.remove();
  document.getElementById('topomap-overlay')?.remove();
  if (window.__topomapCleanup) { try { window.__topomapCleanup(); } catch (e) {} }

  // ---------------------------------------------------------------------
  // Locator: find the heightmap array in WASM linear memory.
  //
  // The site's "Map Generation" quality setting (#qualitySelect: -1 = Low,
  // 0 = Medium, 1 = High) changes the heightmap's resolution: 512x512,
  // 1024x1024, and 2048x2048 respectively (a clean 2x step per level,
  // confirmed empirically - the disc-shaped void/land ratio is identical
  // at all three sizes, only the resolution changes). We read the current
  // setting from the DOM as a hint for which width to try first, but the
  // search itself verifies the result (exact element-count match), so a
  // stale/incorrect hint just costs a wasted pass rather than a wrong map.
  // ---------------------------------------------------------------------
  function qualityWidthHint() {
    const sel = document.getElementById('qualitySelect');
    if (!sel) return null;
    if (sel.value === '-1') return 512;
    if (sel.value === '0') return 1024;
    if (sel.value === '1') return 2048;
    return null;
  }

  function tryLocateWidth(width) {
    const f32 = window.gameInstance.Module.HEAPF32;
    const n = f32.length;
    const W = width * width;
    if (W > n) return null;

    function isNear400(v) { return v >= -401 && v <= -399; }
    function isMain(v) { return v >= -399 && v <= 1000; }
    // observed ratios across all three quality levels: near~0.426, main~0.573;
    // thresholds set well below that so window detection isn't sensitive to
    // exact seed/terrain variation.
    const nearThresh = Math.round(W * 0.35);
    const mainThresh = Math.round(W * 0.45);

    let nearCount = 0, mainCount = 0;
    for (let i = 0; i < W; i++) {
      const v = f32[i];
      if (isNear400(v)) nearCount++;
      if (isMain(v)) mainCount++;
    }

    const validStarts = [];
    for (let start = 0; start + W <= n; start++) {
      if (nearCount >= nearThresh && mainCount >= mainThresh) validStarts.push(start);
      const outV = f32[start];
      if (isNear400(outV)) nearCount--;
      if (isMain(outV)) mainCount--;
      const inIdx = start + W;
      if (inIdx < n) {
        const inV = f32[inIdx];
        if (isNear400(inV)) nearCount++;
        if (isMain(inV)) mainCount++;
      }
    }
    if (!validStarts.length) return null;

    // collapse to contiguous runs of valid window-start positions. A stale
    // heightmap left in memory from a *different* quality level (its array
    // never got overwritten, just abandoned) can be big enough that many
    // small width^2 windows inside it also look statistically valid, so the
    // largest run is not necessarily the real match for this width - check
    // every run (largest first) and keep only the one whose boundary search
    // actually resolves to a region of exactly this width's size.
    const runs = [];
    let rs = validStarts[0], re = validStarts[0];
    for (let k = 1; k < validStarts.length; k++) {
      if (validStarts[k] - re <= 5) { re = validStarts[k]; }
      else { runs.push({ start: rs, end: re }); rs = validStarts[k]; re = validStarts[k]; }
    }
    runs.push({ start: rs, end: re });
    runs.sort((a, b) => (b.end - b.start) - (a.end - a.start));

    const THRESH = 3 * width;
    for (const run of runs) {
      const winStart = Math.round((run.start + run.end) / 2);
      const center = winStart + Math.floor(W / 2);

      let i = center, runLen = 0, inVoidRun = false, topEndpoint = -1;
      while (i >= 0) {
        const v = f32[i];
        if (v === -400) { runLen++; if (runLen >= THRESH) inVoidRun = true; }
        else { runLen = 0; if (inVoidRun) { topEndpoint = i + 1; break; } }
        i--;
      }
      i = center; runLen = 0; inVoidRun = false;
      let bottomEndpoint = -1;
      while (i < n) {
        const v = f32[i];
        if (v === -400) { runLen++; if (runLen >= THRESH) inVoidRun = true; }
        else { runLen = 0; if (inVoidRun) { bottomEndpoint = i - 1; break; } }
        i++;
      }
      if (topEndpoint < 0 || bottomEndpoint < 0) continue;
      const totalLen = bottomEndpoint - topEndpoint + 1;
      // require a tight match - a differently-sized (e.g. stale) region
      // fails this, which is exactly what lets us skip it instead of
      // rendering garbage.
      if (Math.abs(totalLen - W) > 4096) continue;
      return { start: topEndpoint, width, height: width };
    }
    return null;
  }

  function locateHeightmap() {
    const hint = qualityWidthHint();
    const widths = hint ? [hint, ...[512, 1024, 2048].filter(w => w !== hint)] : [1024, 2048, 512];
    for (const width of widths) {
      const loc = tryLocateWidth(width);
      if (loc) return loc;
    }
    throw new Error('topomap: no candidate heightmap region found at any known quality resolution (512/1024/2048)');
  }

  function extractHeightmap() {
    const loc = locateHeightmap();
    const f32 = window.gameInstance.Module.HEAPF32;
    const { start, width, height } = loc;
    // store row-major, already flipped about X-axis to match the app's own orientation
    const data = new Float32Array(width * height);
    let min = Infinity, max = -Infinity;
    for (let row = 0; row < height; row++) {
      const srcRow = height - 1 - row;
      for (let col = 0; col < width; col++) {
        const v = f32[start + srcRow * width + col];
        data[row * width + col] = v;
        if (v > -390) { // exclude void from min/max stats
          if (v < min) min = v;
          if (v > max) max = v;
        }
      }
    }
    return { data, width, height, min, max, loc };
  }

  // ---------------------------------------------------------------------
  // Coloring
  // ---------------------------------------------------------------------
  function lerp(a, b, t) { return a + (b - a) * t; }
  function lerpColor(c1, c2, t) {
    return [lerp(c1[0], c2[0], t), lerp(c1[1], c2[1], t), lerp(c1[2], c2[2], t)];
  }

  function colorForHeight(v, maxHeight) {
    const W = CONFIG.water;
    if (v <= W.voidThreshold) return W.voidColor;
    if (v < W.deepFloor) {
      const t = (v - W.voidThreshold) / (W.deepFloor - W.voidThreshold);
      return lerpColor(W.voidColor, W.deepColor, t);
    }
    if (v < 0) {
      const t = (v - W.deepFloor) / (0 - W.deepFloor);
      return lerpColor(W.deepColor, W.midColor, t);
    }
    if (v < CONFIG.seaLevel) {
      const t = v / CONFIG.seaLevel;
      return lerpColor(W.midColor, W.shallowColor, t);
    }
    if (v >= 150) {
      const maxH = Math.max(maxHeight, 151);
      const t = Math.min(1, (v - 150) / (maxH - 150));
      return lerpColor([255, 255, 255], CONFIG.lightBlueColor, t);
    }
    const stops = CONFIG.landStops;
    for (let s = 0; s < stops.length - 1; s++) {
      const [h0, c0] = stops[s], [h1, c1] = stops[s + 1];
      if (v >= h0 && v <= h1) {
        const t = (v - h0) / (h1 - h0);
        return lerpColor(c0, c1, t);
      }
    }
    return [255, 0, 255]; // shouldn't happen; magenta flags a gap in the stops
  }

  function buildBaseImageData(hm) {
    const { data, width, height, max } = hm;
    const imgData = new ImageData(width, height);
    for (let i = 0; i < width * height; i++) {
      const [r, g, b] = colorForHeight(data[i], max);
      imgData.data[i * 4] = r;
      imgData.data[i * 4 + 1] = g;
      imgData.data[i * 4 + 2] = b;
      imgData.data[i * 4 + 3] = 255;
    }
    return imgData;
  }

  // ---------------------------------------------------------------------
  // Contour lines via marching squares
  // ---------------------------------------------------------------------
  function computeContours(hm, spacing) {
    const { data, width, height, min, max } = hm;
    const levels = [];
    const startK = Math.floor((min - CONFIG.seaLevel) / spacing);
    const endK = Math.ceil((max - CONFIG.seaLevel) / spacing);
    for (let k = startK; k <= endK; k++) {
      const lvl = CONFIG.seaLevel + k * spacing;
      if (lvl > min && lvl < max) levels.push(lvl);
    }

    const segmentsByLevel = new Map();
    const voidThresh = CONFIG.water.voidThreshold;

    for (const level of levels) {
      const segs = [];
      for (let j = 0; j < height - 1; j++) {
        const rowOff = j * width, rowOff1 = (j + 1) * width;
        for (let i = 0; i < width - 1; i++) {
          const v00 = data[rowOff + i], v10 = data[rowOff + i + 1];
          const v01 = data[rowOff1 + i], v11 = data[rowOff1 + i + 1];
          if (v00 <= voidThresh || v10 <= voidThresh || v01 <= voidThresh || v11 <= voidThresh) continue;

          let idx = 0;
          if (v00 >= level) idx |= 1;
          if (v10 >= level) idx |= 2;
          if (v11 >= level) idx |= 4;
          if (v01 >= level) idx |= 8;
          if (idx === 0 || idx === 15) continue;

          const lerpEdge = (va, vb, xa, ya, xb, yb) => {
            const t = (level - va) / (vb - va);
            return [xa + (xb - xa) * t, ya + (yb - ya) * t];
          };
          const top = () => lerpEdge(v00, v10, i, j, i + 1, j);
          const right = () => lerpEdge(v10, v11, i + 1, j, i + 1, j + 1);
          const bottom = () => lerpEdge(v01, v11, i, j + 1, i + 1, j + 1);
          const left = () => lerpEdge(v00, v01, i, j, i, j + 1);

          const pairsByCase = {
            1: [left, top], 2: [top, right], 3: [left, right],
            4: [right, bottom], 5: [top, right, bottom, left], // saddle, draw both diagonally-ish
            6: [top, bottom], 7: [left, bottom],
            8: [left, bottom], 9: [top, bottom],
            10: [top, left, right, bottom], // saddle
            11: [right, bottom], 12: [left, right],
            13: [top, right], 14: [left, top],
          };
          const pts = pairsByCase[idx];
          if (!pts) continue;
          if (idx === 5 || idx === 10) {
            segs.push([pts[0](), pts[1]()]);
            segs.push([pts[2](), pts[3]()]);
          } else {
            segs.push([pts[0](), pts[1]()]);
          }
        }
      }
      segmentsByLevel.set(level, segs);
    }
    return segmentsByLevel;
  }

  const header = overlay.querySelector('#topomap-header');
  const canvas = overlay.querySelector('#topomap-canvas');
  const ctx = canvas.getContext('2d');
  const contourCheck = overlay.querySelector('#topomap-contour-check');
  const spacingButtons = Array.from(overlay.querySelectorAll('.topomap-spacing-btn'));
  const densitySlider = overlay.querySelector('#topomap-density-slider');
  const densityVal = overlay.querySelector('#topomap-density-val');
  const opacitySlider = overlay.querySelector('#topomap-opacity-slider');
  const opacityVal = overlay.querySelector('#topomap-opacity-val');
  const refreshBtn = overlay.querySelector('#topomap-refresh-btn');
  const statusEl = overlay.querySelector('#topomap-status');

  let currentSpacing = CONFIG.contour.defaultSpacing;
  function setSpacing(v) {
    currentSpacing = v;
    spacingButtons.forEach(b => {
      b.style.background = (parseInt(b.dataset.spacing, 10) === v) ? '#3a8a5a' : '#333';
    });
    scheduleDraw();
  }
  spacingButtons.forEach(b => b.addEventListener('click', () => setSpacing(parseInt(b.dataset.spacing, 10))));

  // stop Unity's global key handlers from eating input meant for our controls
  [contourCheck, densitySlider, opacitySlider].forEach(el => {
    ['keydown', 'keyup', 'keypress'].forEach(evt => el.addEventListener(evt, e => e.stopPropagation(), true));
  });

  function densityToPx(density) {
    const [pxLow, pxHigh] = CONFIG.contour.densityToPxRange;
    const t = (density - 1) / 9;
    return pxLow + (pxHigh - pxLow) * t;
  }

  // --- dragging the overlay by its header ---
  let overlayDrag = null;
  header.addEventListener('mousedown', (e) => {
    if (e.target.closest('input, button, select, label')) return; // don't drag when using controls
    const rect = overlay.getBoundingClientRect();
    // lock in explicit px position/size so it stops reacting to vw/vh on next interaction
    overlay.style.left = rect.left + 'px';
    overlay.style.top = rect.top + 'px';
    overlay.style.right = 'auto';
    overlay.style.width = rect.width + 'px';
    overlay.style.height = rect.height + 'px';
    overlayDrag = { startX: e.clientX, startY: e.clientY, startLeft: rect.left, startTop: rect.top };
    e.preventDefault();
  });
  window.addEventListener('mousemove', (e) => {
    if (!overlayDrag) return;
    const dx = e.clientX - overlayDrag.startX;
    const dy = e.clientY - overlayDrag.startY;
    overlay.style.left = (overlayDrag.startLeft + dx) + 'px';
    overlay.style.top = (overlayDrag.startTop + dy) + 'px';
  });
  window.addEventListener('mouseup', () => { overlayDrag = null; });

  let hm = null;
  let baseBitmap = null;
  let contourCache = { spacing: null, segmentsByLevel: null };
  // label positions are computed in grid-space and cached separately from paint
  // (opacity changes must NOT trigger recomputation of this - that was the freeze bug)
  let labelCache = { key: null, positions: null };
  const view = { scale: 1, offsetX: 0, offsetY: 0 };

  let rafPending = false;
  function scheduleDraw() {
    if (rafPending) return;
    rafPending = true;
    requestAnimationFrame(() => { rafPending = false; draw(); });
  }

  function resizeCanvasToContainer() {
    const rect = canvas.parentElement.getBoundingClientRect();
    canvas.width = rect.width;
    canvas.height = rect.height;
  }

  function fitToView() {
    if (!hm) return;
    resizeCanvasToContainer();
    const scale = Math.min(canvas.width / hm.width, canvas.height / hm.height) * 0.95;
    view.scale = scale;
    view.offsetX = (canvas.width - hm.width * scale) / 2;
    view.offsetY = (canvas.height - hm.height * scale) / 2;
  }

  async function rebuildFromMemory() {
    statusEl.textContent = 'Scanning memory...';
    try {
      hm = extractHeightmap();
      const imgData = buildBaseImageData(hm);
      baseBitmap = await createImageBitmap(imgData);
      contourCache = { spacing: null, segmentsByLevel: null };
      fitToView();
      draw();
      statusEl.textContent = `OK (${hm.width}x${hm.height}, max height ${Math.round(hm.max)}m)`;
    } catch (e) {
      statusEl.textContent = 'Error: ' + e.message;
      console.error('[topomap]', e);
    }
  }

  function ensureContours() {
    const spacing = currentSpacing;
    if (contourCache.spacing === spacing && contourCache.segmentsByLevel) return contourCache.segmentsByLevel;
    const segs = computeContours(hm, spacing);
    contourCache = { spacing, segmentsByLevel: segs };
    return segs;
  }

  // Recompute label anchor points (grid-space) only when spacing, density, or
  // zoom bucket actually changes - NOT on every paint. (Recomputing this on every
  // opacity/pan input event was the cause of the freeze: rapid slider drags piled
  // up expensive segment-walking work faster than the main thread could keep up.)
  function ensureLabelPositions(segmentsByLevel, density, scale) {
    const scaleBucket = Math.round(Math.log2(Math.max(scale, 0.01)) * 4);
    const key = `${currentSpacing}|${density}|${scaleBucket}`;
    if (labelCache.key === key) return labelCache.positions;

    const stepPx = Math.max(15, densityToPx(density) / scale);
    const positions = [];
    for (const [level, segs] of segmentsByLevel) {
      let accum = 0;
      for (const seg of segs) {
        const segLen = Math.hypot(seg[1][0] - seg[0][0], seg[1][1] - seg[0][1]);
        accum += segLen;
        if (accum >= stepPx) {
          accum = 0;
          positions.push({ x: (seg[0][0] + seg[1][0]) / 2, y: (seg[0][1] + seg[1][1]) / 2, level });
        }
      }
    }
    labelCache = { key, positions };
    return positions;
  }

  function draw() {
    if (!hm || !baseBitmap) return;
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.fillStyle = '#000';
    ctx.fillRect(0, 0, canvas.width, canvas.height);
    ctx.imageSmoothingEnabled = false;
    ctx.setTransform(view.scale, 0, 0, view.scale, view.offsetX, view.offsetY);
    ctx.drawImage(baseBitmap, 0, 0);

    if (contourCheck.checked) {
      const segmentsByLevel = ensureContours();
      ctx.lineWidth = 1 / view.scale;
      ctx.strokeStyle = CONFIG.contour.lineColor;
      for (const [, segs] of segmentsByLevel) {
        ctx.beginPath();
        segs.forEach((seg) => {
          ctx.moveTo(seg[0][0], seg[0][1]);
          ctx.lineTo(seg[1][0], seg[1][1]);
        });
        ctx.stroke();
      }

      const showLabels = view.scale > 0.9; // skip labels entirely when too zoomed out to read them
      if (showLabels) {
        const density = parseInt(densitySlider.value, 10);
        const labelPositions = ensureLabelPositions(segmentsByLevel, density, view.scale);
        ctx.setTransform(1, 0, 0, 1, 0, 0);
        ctx.font = 'bold 12px monospace';
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.lineWidth = 3;
        ctx.lineJoin = 'round';
        ctx.globalAlpha = parseInt(opacitySlider.value, 10) / 100;
        for (const p of labelPositions) {
          const sx = p.x * view.scale + view.offsetX;
          const sy = p.y * view.scale + view.offsetY;
          if (sx < -20 || sy < -20 || sx > canvas.width + 20 || sy > canvas.height + 20) continue;
          const text = `${p.level}m`;
          ctx.strokeStyle = CONFIG.contour.labelStrokeColor;
          ctx.strokeText(text, sx, sy);
          ctx.fillStyle = CONFIG.contour.labelFillColor;
          ctx.fillText(text, sx, sy);
        }
        ctx.globalAlpha = 1;
      }
    }
  }

  // pan/zoom interaction
  let dragging = false, lastX = 0, lastY = 0;
  canvas.addEventListener('mousedown', (e) => {
    dragging = true; lastX = e.clientX; lastY = e.clientY;
    canvas.style.cursor = 'grabbing';
  });
  window.addEventListener('mouseup', () => { dragging = false; canvas.style.cursor = 'grab'; });
  window.addEventListener('mousemove', (e) => {
    if (!dragging) return;
    view.offsetX += e.clientX - lastX;
    view.offsetY += e.clientY - lastY;
    lastX = e.clientX; lastY = e.clientY;
    scheduleDraw();
  });
  canvas.addEventListener('wheel', (e) => {
    e.preventDefault();
    const rect = canvas.getBoundingClientRect();
    const mx = e.clientX - rect.left, my = e.clientY - rect.top;
    const worldX = (mx - view.offsetX) / view.scale;
    const worldY = (my - view.offsetY) / view.scale;
    const zoomFactor = Math.exp(-e.deltaY * 0.001);
    view.scale = Math.min(50, Math.max(0.1, view.scale * zoomFactor));
    view.offsetX = mx - worldX * view.scale;
    view.offsetY = my - worldY * view.scale;
    scheduleDraw();
  }, { passive: false });

  contourCheck.addEventListener('change', scheduleDraw);
  densitySlider.addEventListener('input', () => {
    densityVal.textContent = densitySlider.value;
    scheduleDraw();
  });
  opacitySlider.addEventListener('input', () => {
    opacityVal.textContent = opacitySlider.value;
    scheduleDraw();
  });
  refreshBtn.addEventListener('click', rebuildFromMemory);

  // Window resize (e.g. browser resized) still re-fits to keep the whole map visible.
  window.addEventListener('resize', () => { if (overlay.style.display !== 'none') { fitToView(); scheduleDraw(); } });

  // Manual resize via the overlay's drag handle: just resize the canvas buffer
  // and redraw, preserving the current pan/zoom instead of re-fitting.
  const resizeObserver = new ResizeObserver(() => {
    if (overlay.style.display === 'none' || !hm) return;
    resizeCanvasToContainer();
    scheduleDraw();
  });
  resizeObserver.observe(overlay);

  let open = false;
  toggleBtn.addEventListener('click', async () => {
    open = !open;
    overlay.style.display = open ? 'flex' : 'none';
    if (open && !hm) {
      // first time loading data: fit the whole map to the view
      await rebuildFromMemory();
    } else if (open) {
      // reopening: keep whatever pan/zoom the user left it at, just
      // resync the canvas backing size in case the window/overlay resized
      resizeCanvasToContainer();
      draw();
    }
  });

  // best-effort auto-refresh: poll document.title (cheap) for seed changes while open
  let lastTitle = document.title;
  const titleWatcher = setInterval(() => {
    if (!open) return;
    if (document.title !== lastTitle) {
      lastTitle = document.title;
      rebuildFromMemory();
    }
  }, 1000);
  window.__topomapCleanup = () => { clearInterval(titleWatcher); resizeObserver.disconnect(); };

  console.log('[topomap] injected. Click the topo icon near the top-right to open.');
})();
